### 1.2 添加随机延迟策略

#### 策略一：智能延迟管理器

```python
import time
import random
from urllib.parse import urlparse

class PoliteRequester:
    """尊重式请求：避免触发速率限制"""
    
    def __init__(self, 
                 min_delay: float = 1.0, 
                 max_delay: float = 3.0):
        self.min_delay = min_delay
        self.max_delay = max_delay
        self.last_request_per_domain = {}
    
    def wait_for_domain(self, url: str):
        """按域名管理延迟，避免跨站干扰"""
        domain = urlparse(url).netloc
        last_request = self.last_request_per_domain.get(domain, 0)
        
        elapsed = time.time() - last_request
        delay = random.uniform(self.min_delay, self.max_delay)
        
        if elapsed < delay:
            time.sleep(delay - elapsed)
        
        self.last_request_per_domain[domain] = time.time()
    
    def human_like_delay(self):
        """模拟人类阅读时间的随机延迟"""
        # 人类阅读速度约200-250词/分钟
        # 平均页面停留时间 2-8秒
        base_delay = random.uniform(2, 8)
        
        # 添加"思考"停顿（10%概率额外停留）
        if random.random() < 0.1:
            base_delay += random.uniform(3, 7)
        
        time.sleep(base_delay)
```

```python
def fetch_with_retry(url: str, retry_count: int = 3):
    """指数退避策略：避免连续失败导致的封禁"""
    for attempt in range(retry_count):
        try:
            response = requests.get(url, timeout=30)
            response.raise_for_status()
            return response
        except requests.RequestException:
            if attempt == retry_count - 1:
                raise
            # 指数退避：2^attempt 秒
            time.sleep(2 ** attempt)
```

import random
import time

class HumanBehaviorSimulator:
    """模拟真实用户行为的爬虫行为"""
    
    def simulate_reading(self, page):
        """模拟页面阅读行为"""
        # 随机滚动
        scroll_amount = random.randint(100, 800)
        page.evaluate(f'window.scrollBy(0, {scroll_amount})')
        
        # 随机停留时间
        stay_time = random.uniform(1.5, 4.0)
        time.sleep(stay_time)
        
        # 10%概率回滚（人类会回头看）
        if random.random() < 0.1:
            page.evaluate(f'window.scrollBy(0, -{random.randint(50, 200)})')
            time.sleep(random.uniform(0.5, 1.5))
    
    def simulate_form_filling(self, page, field_ref, value):
        """模拟人类填写表单"""
        # 先聚焦
        page.focus(field_ref)
        time.sleep(random.uniform(0.1, 0.3))
        
        # 逐字输入（人类打字速度 150-250ms/字符）
        for char in value:
            page.type(field_ref, char)
            time.sleep(random.uniform(0.05, 0.15))
        
        # 填完后检查（人类习惯）
        time.sleep(random.uniform(0.2, 0.5))
    
    def simulate_click_decision(self, page, element_ref):
        """模拟点击前的决策过程"""
        # 悬停查看
        page.hover(element_ref)
        time.sleep(random.uniform(0.3, 0.8))
        
        # 5%概率不点击（人类犹豫）
        if random.random() < 0.05:
            page.hover('body')  # 移开鼠标
            return False
        
        # 执行点击
        page.click(element_ref)
        return True
```

    └── 结果汇总 → 输出

class MultiAgentCoordinator:
    """多代理协调器"""
    
    def launch_parallel_search(self, queries: list[str]):
        """启动并行搜索任务"""
        
        tasks = []
        for i, query in enumerate(queries):
            # 创建子代理任务（使用Task工具）

                prompt=query,
                output_format={

                    "evidence": "string",
                    "source_url": "string",
                    "confidence": "float"
                }
            )
            tasks.append(task)
        
        # 并行执行所有任务

        return self.aggregate_results(results)

        aggregated = {
            'total_findings': len(results),

            'verified_claims': [],
            'sources': []
        }
        
        # 按主题分组
        for result in results:
            topic = result['task_description']
            if topic not in aggregated['by_topic']:
                aggregated['by_topic'][topic] = []
            aggregated['by_topic'][topic].append(result['findings'])
        
        # 交叉验证
        aggregated['verified_claims'] = self.cross_verify(results)
        
        return aggregated
```

│   └── 高敏感网站 → min_delay=3s, max_delay=8s
│
└── 模拟用户行为
    ├── 简单抓取 → 只需随机延迟
    ├── 表单交互 → 添加hover + 逐字输入
    └── 复杂爬取 → 完整行为模拟（滚动、悬停、犹豫）

### 4.3 子代理编排最佳实践

#### 最佳实践清单

✅ **任务分发原则：**

- 任务独立性：避免重叠，明确边界
- 任务粒度：适中（不要太细也不要太粗）

- 附带来源追溯

✅ **结果汇总流程：**
1. 去重处理
2. 交叉验证（≥3源）
3. 信心评分汇总
4. 按重要性排序

✅ **质量检查：**
- 来源多样性 ≥3类型
- 时间跨度 ≥12个月覆盖
- 视角平衡 ≥正反中立

❌ **常见错误：**
- 过多子代理（>10）→ 协调成本高
- 任务重叠 → 重复工作
- 输出格式不一致 → 合成困难
- 缺少延迟策略 → 触发封禁
