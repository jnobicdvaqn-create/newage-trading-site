```python
import pandas as pd
import numpy as np

def double_ma_strategy(data, short_window=10, long_window=30):
    """
    基础双均线策略
```

#### 优化方法1：MACD过滤（趋势确认）

**原理：** MACD判断大趋势方向，只在趋势方向交易

```python
def ma_with_macd_filter(data, short_window=10, long_window=30):
    """
    双均线 + MACD趋势过滤
    只在MACD同向时交易
    """
    # 双均线
    data['short_ma'] = data['close'].rolling(window=short_window).mean()
    data['long_ma'] = data['close'].rolling(window=long_window).mean()
    
    # MACD指标
    exp12 = data['close'].ewm(span=12, adjust=False).mean()
    exp26 = data['close'].ewm(span=26, adjust=False).mean()
    data['macd'] = exp12 - exp26
    data['macd_signal'] = data['macd'].ewm(span=9, adjust=False).mean()
    data['macd_hist'] = data['macd'] - data['macd_signal']
    
    # 生成信号（增加MACD过滤）
    data['signal'] = 0
    
    # 做多条件：短均>长均 且 MACD>0（上升趋势）
    long_condition = (data['short_ma'] > data['long_ma']) & (data['macd_hist'] > 0)
    data.loc[long_condition, 'signal'] = 1
    
    # 做空条件：短均<长均 且 MACD<0（下降趋势）
    short_condition = (data['short_ma'] < data['long_ma']) & (data['macd_hist'] < 0)
    data.loc[short_condition, 'signal'] = -1
    
    return data
```

**效果：** 
- 减少震荡市场假信号
- 提高胜率（通常+5-10%）
- 但可能降低交易频率

#### 优化方法2：RSI过滤（超买超卖过滤）

**原理：** RSI判断超买超卖，避免极端位置入场

```python
def ma_with_rsi_filter(data, short_window=10, long_window=30, 
                       rsi_period=14, rsi_upper=70, rsi_lower=30):
    """
    双均线 + RSI超买超卖过滤
    避免在极端位置入场
    """
    # 双均线
    data['short_ma'] = data['close'].rolling(window=short_window).mean()
    data['long_ma'] = data['close'].rolling(window=long_window).mean()
    
    # RSI计算
    delta = data['close'].diff()
    gain = (delta.where(delta > 0, 0)).rolling(window=rsi_period).mean()
    loss = (-delta.where(delta < 0, 0)).rolling(window=rsi_period).mean()
    rs = gain / loss
    data['rsi'] = 100 - (100 / (1 + rs))
    
    # 生成信号（增加RSI过滤）
    data['signal'] = 0
    
    # 做多：均线金叉 且 RSI<70（不在超买区）
    long_condition = (data['short_ma'] > data['long_ma']) & (data['rsi'] < rsi_upper)
    data.loc[long_condition, 'signal'] = 1
    
    # 做空：均线死叉 且 RSI>30（不在超卖区）
    short_condition = (data['short_ma'] < data['long_ma']) & (data['rsi'] > rsi_lower)
    data.loc[short_condition, 'signal'] = -1
    
    return data
```

**效果：**
- 避免追高杀低
- 提高盈亏比
- 适合趋势性品种

#### 优化方法3：成交量确认

**原理：** 放量突破才入场，确认趋势强度

```python
def ma_with_volume_confirmation(data, short_window=10, long_window=30,
                                volume_threshold=1.5):
    """
    双均线 + 成交量确认
    只在放量突破时交易
    """
    # 双均线
    data['short_ma'] = data['close'].rolling(window=short_window).mean()
    data['long_ma'] = data['close'].rolling(window=long_window).mean()
    
    # 成交量指标
    data['volume_ma'] = data['volume'].rolling(window=20).mean()
    data['volume_ratio'] = data['volume'] / data['volume_ma']
    
    # 均线交叉点
    data['cross_up'] = (data['short_ma'] > data['long_ma']) & \
                        (data['short_ma'].shift(1) <= data['long_ma'].shift(1))
    data['cross_down'] = (data['short_ma'] < data['long_ma']) & \
                          (data['short_ma'].shift(1) >= data['long_ma'].shift(1))
    
    # 生成信号（需要成交量确认）
    data['signal'] = 0
    
    # 做多：金叉 + 放量
    long_condition = data['cross_up'] & (data['volume_ratio'] > volume_threshold)
    data.loc[long_condition, 'signal'] = 1
    
    # 做空：死叉 + 放量
    short_condition = data['cross_down'] & (data['volume_ratio'] > volume_threshold)
    data.loc[short_condition, 'signal'] = -1
    
    # 持仓状态延续（非交叉日保持仓位）
    data['signal'] = data['signal'].replace(to_replace=0, method='ffill')
    
    return data
```

**效果：**
- 提高信号质量
- 减少假突破
- 但可能错过部分机会

### 1.2 多因子策略入门

#### 三因子模型（趋势+动量+波动率）

```python
import talib

def multi_factor_strategy(data):
    """
    多因子策略：趋势 + 动量 + 波动率
    三个因子综合评分后决策
    """
    # ========== 因子1：趋势（ADX） ==========
    data['adx'] = talib.ADX(data['high'], data['low'], data['close'], timeperiod=14)
    # ADX>25表示趋势明显
    data['trend_factor'] = np.where(data['adx'] > 25, 1, 0)
    
    # ========== 因子2：动量（RSI） ==========
    data['rsi'] = talib.RSI(data['close'], timeperiod=14)
    # RSI 50-70多头，30-50空头
    data['momentum_factor'] = np.where(
        (data['rsi'] > 50) & (data['rsi'] < 70), 1,
        np.where((data['rsi'] < 50) & (data['rsi'] > 30), -1, 0)
    )
    
    # ========== 因子3：波动率（ATR） ==========
    data['atr'] = talib.ATR(data['high'], data['low'], data['close'], timeperiod=14)
    data['atr_ma'] = data['atr'].rolling(window=20).mean()
    # ATR高于均线表示波动率放大，趋势机会
    data['volatility_factor'] = np.where(data['atr'] > data['atr_ma'], 1, 0)
    
    # ========== 综合评分 ==========
    data['total_score'] = (
        data['trend_factor'] * 0.4 +     # 趋势权重40%
        data['momentum_factor'] * 0.4 +  # 动量权重40%
        data['volatility_factor'] * 0.2  # 波动率权重20%
    )
    
    # ========== 生成信号 ==========
    data['signal'] = 0
    data.loc[data['total_score'] > 0.5, 'signal'] = 1   # 多头信号
    data.loc[data['total_score'] < -0.5, 'signal'] = -1 # 空头信号
    
    return data
```

**多因子策略优势：**
- 多维度验证，提高信号质量
- 降低单一指标失效风险
- 可根据市场状态调整因子权重

```python
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import classification_report

def ml_trading_strategy(data, lookback=20):
    """
    机器学习交易策略（随机森林）
    使用历史特征预测未来涨跌
    """
    # ========== 特征工程 ==========
    features = pd.DataFrame()
    
    # 技术指标特征
    features['return_1d'] = data['close'].pct_change(1)
    features['return_5d'] = data['close'].pct_change(5)
    features['return_10d'] = data['close'].pct_change(10)
    features['volatility'] = data['close'].pct_change().rolling(lookback).std()
    features['rsi'] = talib.RSI(data['close'], timeperiod=14)
    features['adx'] = talib.ADX(data['high'], data['low'], data['close'], timeperiod=14)
    
    # 价格位置特征
    features['price_to_ma10'] = data['close'] / data['close'].rolling(10).mean()
    features['price_to_ma30'] = data['close'] / data['close'].rolling(30).mean()
    features['price_to_high'] = data['close'] / data['high'].rolling(lookback).max()
    features['price_to_low'] = data['close'] / data['low'].rolling(lookback).min()
    
    # ========== 标签构造 ==========
    # 未来N日收益率
    future_return = data['close'].shift(-5) / data['close'] - 1
    labels = np.where(future_return > 0.02, 1,      # 上涨>2%
                     np.where(future_return < -0.02, -1, 0))  # 下跌>2%
    
    # ========== 模型训练 ==========
    features = features.dropna()
    labels = labels[:len(features)]
    
    X_train, X_test, y_train, y_test = train_test_split(
        features[:-100], labels[:-100], test_size=0.2, shuffle=False
    )
    
    model = RandomForestClassifier(
        n_estimators=100,
        max_depth=5,
        min_samples_split=10,
        random_state=42
    )
    model.fit(X_train, y_train)
    
    # 模型评估
    y_pred = model.predict(X_test)
    print(classification_report(y_test, y_pred))
    
    # ========== 生成信号 ==========
    data['ml_signal'] = 0
    data.loc[features.index, 'ml_signal'] = model.predict(features)
    
    return data, model
```

**机器学习注意事项：**
1. **避免过拟合：** 使用样本外数据测试
2. **特征选择：** 特征不宜过多（<20个）
3. **模型简单：** 首选线性模型、随机森林
4. **定期重训练：** 市场变化后需更新模型

### 2.1 仓位管理方法

#### 方法1：固定比例法

**最简单实用的方法**

```python
def fixed_fraction_position(capital, risk_per_trade=0.02, stop_loss_pct=0.05):
    """
    固定比例仓位管理
    每次交易风险不超过总资金的固定比例
    
    参数：
    - capital: 总资金
    - risk_per_trade: 单笔风险比例（默认2%）
    - stop_loss_pct: 止损幅度（默认5%）
    
    返回：建议仓位金额
    """
    risk_amount = capital * risk_per_trade  # 单笔最大亏损金额
    position_size = risk_amount / stop_loss_pct  # 反推仓位
    
    return min(position_size, capital * 0.3)  # 上限30%仓位
```

**示例：**
- 总资金：100万
- 单笔风险：2万（2%）
- 止损：5%
- 仓位：2万/5% = 40万

**原则：**
- 单笔风险不超过总资金2%
- 单一品种仓位不超过30%
- 总仓位不超过80%

#### 方法2：凯利公式

**理论上最优，但实战需谨慎**

```python
def kelly_position(win_rate, avg_win, avg_loss, kelly_fraction=0.5):
    """
    凯利公式仓位计算
    
    公式：f* = (p*b - q) / b
    其中：
    - p = 胜率
    - q = 1 - p（败率）
    - b = 平均盈利/平均亏损（盈亏比）
    
    参数：
    - win_rate: 历史胜率
    - avg_win: 平均盈利比例
    - avg_loss: 平均亏损比例
    - kelly_fraction: 凯利系数（保守起见用0.25-0.5）
    
    返回：建议仓位比例
    """
    b = avg_win / avg_loss  # 盈亏比
    p = win_rate
    q = 1 - win_rate
    
    # 凯利公式
    kelly_pct = (p * b - q) / b
    
    # 保守调整（使用半凯利）
    safe_kelly = kelly_pct * kelly_fraction
    
    # 限制在合理范围内
    return max(0, min(safe_kelly, 0.25))  # 上限25%

# 示例计算
# 假设策略：胜率55%，平均盈利8%，平均亏损5%
position = kelly_position(win_rate=0.55, avg_win=0.08, avg_loss=0.05)
print(f"建议仓位：{position:.1%}")  # 约16.3%
```

**实战建议：**
- 使用**半凯利或四分之一凯利**
- 定期重新计算（每月/每季度）
- 新策略先用固定比例，稳定后再用凯利

#### 方法3：ATR动态仓位

**根据波动率调整仓位大小**

```python
def atr_position(capital, price, atr, risk_multiple=2, risk_per_trade=0.02):
    """
    ATR仓位管理
    波动率大时减小仓位，波动率小时增大仓位
    
    参数：
    - capital: 总资金
    - price: 当前价格
    - atr: 当前ATR值
    - risk_multiple: ATR倍数（默认2倍ATR止损）
    - risk_per_trade: 单笔风险比例（默认2%）
    
    返回：建议手数
    """
    risk_amount = capital * risk_per_trade  # 单笔风险金额
    stop_loss_amount = atr * risk_multiple  # 止损金额（每单位）
    
    # 计算手数
    units = risk_amount / stop_loss_amount
    
    return int(units)  # 返回整数手数

# 示例
# 资金100万，价格50元，ATR=2元
# 止损=2*2=4元，单笔风险2万
# 手数 = 20000/4 = 5000股
```

**ATR仓位优势：**
- 自动适应市场波动
- 波动大时仓位小，风险可控
- 波动小时仓位大，充分利用机会

#### 策略2：移动止损（追踪止损）

```python
def trailing_stop_loss(data, trail_pct=0.05):
    """
    移动止损
    价格上涨时止损线上移，下跌时保持
    
    优点：保护利润，让利润奔跑
    """
    data['highest'] = data['close'].expanding().max()  # 持仓期间最高价
    data['trailing_stop'] = data['highest'] * (1 - trail_pct)
    
    # 止损触发
    data['stop_triggered'] = data['close'] < data['trailing_stop']
    
    return data
```

**组合止损策略：**

```python
def combined_exit_strategy(data, entry_idx, entry_price):
    """
    组合止损：固定止损+移动止损+时间止损
    优先级：固定止损 > 移动止损 > 时间止损
    """
    # 初始化止损价格
    stop_price = entry_price * 0.95  # 固定止损5%
    highest_price = entry_price
    max_days = 20
    current_idx = entry_idx
    
    while current_idx < len(data):
        current_price = data.loc[current_idx, 'close']
        atr = data.loc[current_idx, 'atr']
        
        # 更新最高价
        if current_price > highest_price:
            highest_price = current_price
        
        # 检查固定止损
        if current_price <= entry_price * 0.95:
            return current_idx, 'fixed_stop'
        
        # 检查移动止损（盈利后启用）
        if highest_price > entry_price * 1.03:  # 盈利3%后启用移动止损
            trailing_stop = highest_price * 0.97
            if current_price < trailing_stop:
                return current_idx, 'trailing_stop'
        
        # 检查时间止损
        if current_idx - entry_idx >= max_days:
            return current_idx, 'time_stop'
        
        current_idx += 1
    
    return len(data) - 1, 'end_of_data'
```

### 2.3 最大回撤控制方法

```python
def drawdown_based_position(capital, peak_capital, base_position=0.3):
    """
    根据回撤调整仓位
    回撤越大，仓位越小
    """
    drawdown = (peak_capital - capital) / peak_capital
    
    if drawdown < 0.05:  # 回撤<5%
        position = base_position
    elif drawdown < 0.10:  # 回撤5-10%
        position = base_position * 0.7
    elif drawdown < 0.15:  # 回撤10-15%
        position = base_position * 0.5
    elif drawdown < 0.20:  # 回撤15-20%
        position = base_position * 0.3
    else:  # 回撤>20%，暂停交易
        position = 0
    
    return position
```

#### 方法2：熔断机制

```python
class CircuitBreaker:
    """
    回撤熔断机制
    达到阈值后暂停交易
    """
    def __init__(self, daily_limit=0.03, weekly_limit=0.05, monthly_limit=0.10):
        self.daily_limit = daily_limit
        self.weekly_limit = weekly_limit
        self.monthly_limit = monthly_limit
        
    def check_circuit_breaker(self, daily_dd, weekly_dd, monthly_dd):
        """
        检查是否触发熔断
        返回：是否允许交易，熔断类型
        """
        if monthly_dd >= self.monthly_limit:
            return False, 'monthly_circuit'
        elif weekly_dd >= self.weekly_limit:
            return False, 'weekly_circuit'
        elif daily_dd >= self.daily_limit:
            return False, 'daily_circuit'
        else:
            return True, None
```

#### 方法3：风险预算

```python
def risk_budget_allocation(total_capital, strategies):
    """
    风险预算分配
    根据策略风险分配资金
    """
    total_risk = 0
    risk_budgets = {}
    
    for strategy, params in strategies.items():
        # 每个策略的最大风险
        max_dd = params['max_drawdown']
        risk_budget = total_capital * max_dd
        risk_budgets[strategy] = risk_budget
        total_risk += risk_budget
    
    # 确保总风险不超过总资金的限制
    max_total_risk = total_capital * 0.20  # 总风险不超过20%
    
    if total_risk > max_total_risk:
        # 按比例缩减
        scale = max_total_risk / total_risk
        risk_budgets = {k: v * scale for k, v in risk_budgets.items()}
    
    return risk_budgets
```

```python
def stress_test_scenario(returns, scenarios):
    """
    场景压力测试
    模拟极端市场情况下的策略表现
    
    scenarios: {
        'flash_crash': -0.20,  # 闪崩
        'black_swan': -0.35,   # 黑天鹅
        'volatility_spike': 3.0,  # 波动率飙升
    }
    """
    results = {}
    
    for scenario_name, shock in scenarios.items():
        if 'volatility' in scenario_name:
            # 波动率冲击
            stressed_returns = returns * shock
        else:
            # 价格冲击
            stressed_returns = returns.copy()
            stressed_returns[-5:] = shock  # 最近5天冲击
        
        # 计算压力下的指标
        cumulative_return = (1 + stressed_returns).prod() - 1
        max_dd = calculate_max_drawdown(stressed_returns)
        var_95 = np.percentile(stressed_returns, 5)

            'cumulative_return': cumulative_return,
            'max_drawdown': max_dd,
            'var_95': var_95
        }
    
    return results

def calculate_max_drawdown(returns):
    """计算最大回撤"""
    cumulative = (1 + returns).cumprod()
    running_max = cumulative.cummax()
    drawdown = (cumulative - running_max) / running_max
    return drawdown.min()
```

#### 蒙特卡洛压力测试

```python
def monte_carlo_stress_test(returns, n_simulations=10000, n_days=252):
    """
    蒙特卡洛压力测试
    生成大量随机路径，评估极端情况概率
    """
    mu = returns.mean()
    sigma = returns.std()
    
    # 模拟路径
    simulated_paths = np.zeros((n_simulations, n_days))
    
    for i in range(n_simulations):
        daily_returns = np.random.normal(mu, sigma, n_days)
        simulated_paths[i] = (1 + daily_returns).cumprod()
    
    # 计算统计量
    final_returns = simulated_paths[:, -1] - 1
    max_drawdowns = []
    
    for path in simulated_paths:
        peak = np.maximum.accumulate(path)
        dd = (path - peak) / peak
        max_drawdowns.append(dd.min())
    
    return {
        'expected_return': np.mean(final_returns),
        'worst_case_return': np.percentile(final_returns, 5),
        'expected_max_dd': np.mean(max_drawdowns),
        'worst_case_max_dd': np.percentile(max_drawdowns, 95),
        'prob_loss': np.sum(final_returns < 0) / n_simulations
    }
```

### 3.1 如何避免过拟合

#### 过拟合的信号

| 症状 | 可能原因 |
|------|---------|
| 回测年化>50% | 参数过度优化 |
| 样本外表现断崖下跌 | 模型记忆噪声 |
| 每次改参数都提升 | 在"拟合"数据而非"学习"规律 |
| 交易规则过于复杂 | 对历史数据过度拟合 |

#### 防止过拟合的方法

**1. 样本内/样本外测试**

```python
def in_out_sample_test(data, train_ratio=0.7):
    """
    样本内/样本外测试
    用70%数据训练参数，30%数据验证
    """
    split_idx = int(len(data) * train_ratio)
    
    in_sample = data.iloc[:split_idx]
    out_sample = data.iloc[split_idx:]
    
    # 在样本内优化参数
    best_params = optimize_parameters(in_sample)
    
    # 用最佳参数在样本外测试
    out_sample_results = backtest(out_sample, best_params)
    in_sample_results = backtest(in_sample, best_params)
    
    # 比较表现
    degradation = (in_sample_results['sharpe'] - out_sample_results['sharpe']) / in_sample_results['sharpe']
    
    print(f"样本内夏普: {in_sample_results['sharpe']:.2f}")
    print(f"样本外夏普: {out_sample_results['sharpe']:.2f}")
    print(f"性能衰减: {degradation:.1%}")
    
    # 衰减超过30%视为过拟合
    if degradation > 0.3:
        print("⚠️ 警告：可能过拟合！")
    
    return in_sample_results, out_sample_results
```

**2. 滚动窗口回测（Walk-Forward Analysis）**

```python
def walk_forward_analysis(data, window_size=252, step_size=63, n_windows=10):
    """
    滚动窗口回测
    每次用过去一年数据训练，未来一季度测试
    然后滚动窗口，重复过程
    """
    results = []
    
    for i in range(n_windows):
        # 训练窗口
        train_start = i * step_size
        train_end = train_start + window_size
        
        # 测试窗口
        test_start = train_end
        test_end = test_start + step_size
        
        if test_end > len(data):
            break
        
        train_data = data.iloc[train_start:train_end]
        test_data = data.iloc[test_start:test_end]
        
        # 在训练窗口优化参数
        params = optimize_parameters(train_data)
        
        # 在测试窗口验证
        test_result = backtest(test_data, params)
        results.append(test_result)
    
    # 汇总结果
    avg_return = np.mean([r['annual_return'] for r in results])
    avg_sharpe = np.mean([r['sharpe'] for r in results])
    avg_dd = np.mean([r['max_dd'] for r in results])
    
    print(f"滚动窗口平均年化收益: {avg_return:.2%}")
    print(f"滚动窗口平均夏普: {avg_sharpe:.2f}")
    print(f"滚动窗口平均最大回撤: {avg_dd:.2%}")
    
    return results
```

```python
def parameter_stability_test(data, param_name, param_range, base_params):
    """
    参数稳定性测试
    检查参数在合理范围内是否表现稳定
    """
    results = []
    
    for param_value in param_range:
        params = base_params.copy()
        params[param_name] = param_value
        
        result = backtest(data, params)
        results.append({
            'param_value': param_value,
            'sharpe': result['sharpe'],
            'return': result['annual_return'],
            'max_dd': result['max_dd']
        })
    
    # 分析稳定性
    sharpes = [r['sharpe'] for r in results]
    sharpe_std = np.std(sharpes)
    
    print(f"参数{param_name}在范围内夏普比率标准差: {sharpe_std:.2f}")
    
    if sharpe_std > 0.3:
        print(f"⚠️ 参数{param_name}不稳定，可能过拟合")
    else:
        print(f"✅ 参数{param_name}相对稳定")
    
    return results
```

### 3.2 回测评估指标

```python
def evaluate_backtest(returns, benchmark_returns=None):
    """
    全面的回测评估
    """
    metrics = {}
    
    # ========== 基础指标 ==========
    metrics['total_return'] = (1 + returns).prod() - 1
    metrics['annual_return'] = (1 + metrics['total_return']) ** (252 / len(returns)) - 1
    metrics['volatility'] = returns.std() * np.sqrt(252)
    
    # ========== 风险指标 ==========
    cumulative = (1 + returns).cumprod()
    peak = cumulative.cummax()
    drawdown = (cumulative - peak) / peak
    metrics['max_drawdown'] = drawdown.min()
    
    # ========== 风险调整收益 ==========
    metrics['sharpe_ratio'] = metrics['annual_return'] / metrics['volatility']
    metrics['sortino_ratio'] = metrics['annual_return'] / (returns[returns < 0].std() * np.sqrt(252))
    metrics['calmar_ratio'] = metrics['annual_return'] / abs(metrics['max_drawdown'])
    
    # ========== 交易指标 ==========
    trades = returns[returns != 0]
    metrics['trade_count'] = len(trades)
    metrics['win_rate'] = (trades > 0).sum() / len(trades) if len(trades) > 0 else 0
    metrics['avg_win'] = trades[trades > 0].mean() if (trades > 0).any() else 0
    metrics['avg_loss'] = abs(trades[trades < 0].mean()) if (trades < 0).any() else 0
    metrics['profit_factor'] = metrics['avg_win'] / metrics['avg_loss'] if metrics['avg_loss'] > 0 else np.inf
    
    # ========== 相对指标 ==========
    if benchmark_returns is not None:
        excess_return = returns - benchmark_returns
        metrics['alpha'] = (1 + excess_return).prod() - 1
        metrics['beta'] = returns.cov(benchmark_returns) / benchmark_returns.var()
        metrics['information_ratio'] = excess_return.mean() / excess_return.std() * np.sqrt(252)
    
    return metrics
```

```python
def strategy_failure_detection(metrics_history):
    """
    策略失效检测
    基于历史指标判断策略是否需要调整
    """
    warnings = []
    
    # 1. 连续亏损
    recent_returns = metrics_history['returns'][-20:]  # 最近20个交易日
    if sum(r < 0 for r in recent_returns) >= 15:
        warnings.append("⚠️ 连续亏损预警：最近20天中有15天亏损")
    
    # 2. 夏普比率下降
    recent_sharpe = np.mean(metrics_history['sharpe'][-10:])
    historical_sharpe = np.mean(metrics_history['sharpe'][:-10])
    if recent_sharpe < historical_sharpe * 0.5:
        warnings.append(f"⚠️ 夏普比率下降：从{historical_sharpe:.2f}降至{recent_sharpe:.2f}")
    
    # 3. 胜率下降
    recent_win_rate = metrics_history['win_rate'][-10:].mean()
    historical_win_rate = metrics_history['win_rate'][:-10].mean()
    if recent_win_rate < historical_win_rate * 0.7:
        warnings.append(f"⚠️ 胜率下降：从{historical_win_rate:.1%}降至{recent_win_rate:.1%}")
    
    # 4. 最大回撤突破
    current_dd = metrics_history['max_dd'][-1]
    if current_dd < -0.15:
        warnings.append(f"⚠️ 最大回撤突破15%：{current_dd:.1%}")
    
    # 5. 与基准相关性异常
    recent_corr = metrics_history['benchmark_corr'][-10:].mean()
    if recent_corr > 0.9:
        warnings.append("⚠️ 与基准高度相关，策略可能失效")
    
    return warnings
```
